400 REM Check if number n can be placed at (row, col)
410 DEF FN isSafe(row, col, n) AS INTEGER
420 LOCAL i, j, boxRow, boxCol
440 REM Check row
450 FOR i = 1 TO 9
460 IF s(row, i) = n THEN RETURN 0
470 NEXT i
490 REM Check column
500 FOR i = 1 TO 9
510 IF s(i, col) = n THEN RETURN 0
520 NEXT i
540 REM Check 3x3 box
550 boxRow = INT((row - 1) / 3) * 3 + 1
560 boxCol = INT((col - 1) / 3) * 3 + 1
570 FOR i = boxRow TO boxRow + 2
580 FOR j = boxCol TO boxCol + 2
590 IF s(i, j) = n THEN RETURN 0
600 NEXT j
610 NEXT i
630 RETURN 1
640 END DEF
700 REM Find empty cell (with 0) and return row, col
710 REM Returns 1 if found, 0 if no empty cell
720 DEF FN findEmpty(row, col) AS INTEGER
730 LOCAL i, j
750 FOR i = 1 TO 9
760 FOR j = 1 TO 9
770 IF s(i, j) = 0 THEN
780 LET row = i
790 LET col = j
800 RETURN 1
810 END IF
820 NEXT j
830 NEXT i
850 RETURN 0
860 END DEF
900 REM Main solving function using backtracking
910 DEF FN solve() AS INTEGER
920 LOCAL row, col, num, found
940 REM Find empty cell
950 found = FN findEmpty(row, col)
960 IF found = 0 THEN RETURN 1  REM No empty cell = solved!
980 REM Try numbers 1-9
990 FOR num = 1 TO 9
1000 IF FN isSafe(row, col, num) = 1 THEN
1010 LET s(row, col) = num
1030 REM Recursively try to solve
1040 IF FN solve() = 1 THEN RETURN 1
1060 REM Backtrack
1070 LET s(row, col) = 0
1080 END IF
1090 NEXT num
1110 RETURN 0  REM Trigger backtracking
1120 END DEF

3030 DEF FN verify() AS INTEGER
3040 LOCAL i, j, k, boxRow, boxCol, count
3060 REM Check all rows
3070 FOR i = 1 TO 9
3080 FOR k = 1 TO 9
3090 count = 0

3110 IF s(i, j) = k THEN count = count + 1
3120 NEXT j
3130 IF count <> 1 THEN RETURN 0
3140 NEXT k
3150 NEXT i
3170 REM Check all columns
3180 FOR j = 1 TO 9
3190 FOR k = 1 TO 9
3200 count = 0
3210 FOR i = 1 TO 9
3220 IF s(i, j) = k THEN count = count + 1
3230 NEXT i
3240 IF count <> 1 THEN RETURN 0
3250 NEXT k
3260 NEXT j
3280 REM Check all 3x3 boxes
3290 FOR boxRow = 1 TO 9 STEP 3
3300 FOR boxCol = 1 TO 9 STEP 3
3310 FOR k = 1 TO 9
3320 count = 0
3330 FOR i = boxRow TO boxRow + 2
3340 FOR j = boxCol TO boxCol + 2
3350 IF s(i, j) = k THEN count = count + 1
3360 NEXT j
3370 NEXT i
3380 IF count <> 1 THEN RETURN 0
3390 NEXT k
3400 NEXT boxCol
3410 NEXT boxRow
3430 RETURN 1
3440 END DEF
